% DSDE Workshop Handout PDF Formatting
%
% This LaTeX script is designed to format PDF handouts for the DSDE Workshop.
% It ensures consistent styling, fonts, and layouts throughout the document.
%
% The script should be included in the YAML header of Quarto (.qmd) documents
% structured within different workshop directories. The script manages:
%
% - Configuring global settings and packages for geometry, color, fonts, etc.
% - Defining custom headers and footers with dynamic content
% - Setting custom title page, section fonts, and callout boxes
% - Including specific images and backgrounds
%
% To use, reference this LaTeX script in the YAML header of Quarto documents
% with correct relative paths based on their directory depth, and ensure 
% that the "rootdir" and "stylesdir" paths in this LaTeX script match those 
% relative paths. Adjust as necessary.
%
% Examples:
% For Quarto document located at "Workshops/proj1.qmd":
% "format:
%   pdf:
%     include-in-header:
%       - text: |
%           \input{../Code/Handout Formatting/PDF Formatting Script.tex}"
%
% For Quarto document located at "Workshops/Git-GitHub/Code/proj2.qmd":
% "format:
%   pdf:
%     include-in-header: 
%       - text: |
%           \input{../../../../Code/Handout Formatting/PDF Formatting Script.tex}"
%
% Author: Shelby Golden, M.S. (with assistance from Yale's AI Clarity)
% Date: September 23rd, 2025

% Define Backgrounds
% Background for the first page
\newcommand{\SetFirstPageBackground}{%
  \AddToShipoutPictureBG*{%
    \put(0,0){%
      \includegraphics[width=\paperwidth,height=\paperheight]{\rootdir/Background_Binary Code.png}%
    }
  }
}

% Background for other pages
\newcommand{\SetOtherPagesBackground}{%
  \AddToShipoutPictureBG*{%
    \put(0,0){%
      \includegraphics[width=\paperwidth,height=\paperheight]{\rootdir/Background.png}%
    }
  }
}

% Ensure correct header/footer setup for all pages
\AtBeginDocument{%
  \AtBeginShipoutFirst{%
    \SetFirstPageBackground
  }
  \AtBeginShipout{%
    \ifnum\value{page}>1
    \SetOtherPagesBackground
    \fi
  }
}

% Define font for title, subtitle, author, and date
% Ensure the specified path correctly points to the fonts directory
\newcommand{\customtitlefont}{%
  \fontspec[
    Path=\stylesdir/, % Correctly set the path to the font directory
    UprightFont=*-Roman,
    BoldFont=*-Bold,
    ItalicFont=*-Italic,
    BoldItalicFont=*-BoldItalic
  ]{YaleNew}\fontsize{30}{35}\selectfont\color{DSDEBlue}\bfseries
}

\newcommand{\customsubtitlefont}{\fontspec{Barlow}\fontsize{15}{14}\selectfont\color{DSDEBlue}}
\newcommand{\customauthorfont}{\fontspec{Barlow}\fontsize{11}{0}\selectfont\color{DSDEBlue}\itshape}
\newcommand{\customdatefont}{\fontspec{Barlow}\fontsize{11}{0}\selectfont\color{DSDEBlue}\itshape}

% Define fonts for sectioning commands
\newfontfamily\sectionfont[Color=DSDEBlue]{YaleNew}

\addtokomafont{section}{\sectionfont}
\addtokomafont{subsection}{\subsectionfont}
\addtokomafont{subsubsection}{\subsubsectionfont}

% Set main font for normal text
\setmainfont{Barlow}
\renewcommand{\normalsize}{\fontsize{11}{14}\selectfont\color{DSDEBlue}}
\setlength{\parskip}{1em}

% Set title page formatting
\makeatletter
\def\@maketitle{%
  \newpage
  \null
  \begin{center}%
  {\customtitlefont\@title\par}%
  \ifx\@subtitle\@empty\else%
  \vskip 2em%
  {\customsubtitlefont\@subtitle\par}%
  \fi%
  \ifx\@author\@empty\else%
  \vskip 1em%
  {\customauthorfont\@author\par}%
  \fi%
  \ifx\b@date\@empty\else%
  \vskip 1em%
  {\customdatefont\@date\par}%
  \fi%
  \end{center}%
  \vskip 1em%
}
\makeatother
